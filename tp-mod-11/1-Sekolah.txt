program Sekolah
kamus (global)
    constant NMAX : integer = 1001
    type tabMurid : array [1..NMAX] of string
kamus (lokal)
    N, idxRes : integer
    target : string
    daftarMurid : tabMurid
algoritma
    input(N)
    inputMurid(daftarMurid, N)
    input(target)
    idxRes <- cariMuridBinary(daftarMurid, N, target)
    if idxRes == -1 then
        output("Murid tak terdaftar")
    else
        output("Murid terdaftar dan berada pada absen ke-", idxRes)
    endif
endprogram



procedure inputMurid(in/out A : tabMurid, in n : integer)
{
    IS. terdefinisi sembarang A, terdefinisi n
    FS. A berisi data sebanyak n
}
kamus
    i : integer
algoritma
    for i <- 1 to n do
        input(A[i])
    endfor
endprocedure



function cariMuridBinary(A : tabMurid, n : integer, x : string) -> integer
{ mengembalikan index nilai yg dicari, -1 jika tidak ada. }
kamus
    left, right, mid : integer
    found : integer
algoritma
    left <- 1
    right <- n
    found <- -1
    while left <= right and found == -1 do
        mid <- (left + right) div 2
        if A[mid] == x then
            found <- mid
        else if x < A[mid] then
			right <- mid - 1
        else if x > A[mid] then
			left <- mid + 1
        endif
    endwhile
    return found
endfunction
