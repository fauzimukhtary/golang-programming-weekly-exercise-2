program PolisiPatroli
kamus (global)

    type kota <
        nama : string
        penduduk : integer
        laporan : real
    >

    type tabKota : array [1..20] of kota

kamus (lokal)

    kota : tabKota
    i, jlhKota, populasiAwal : integer

algoritma

    input(jlhKota, populasiAwal)

    for i <- 1 to jlhKota do
        input(kota[i].nama)
        kota[i].penduduk <- populasiAwal div 10
        populasiAwal <- populasiAwal - kota[i].penduduk
        kota[i].laporan <- (kota[i].penduduk mod 1000) * 0.025
    endfor

    for i <- 1 to jlhKota do
        if kota[i].laporan > 5.0 then
            output(kota[i].nama, "dengan tingkat kejahatan:", kota[i].laporan, "%.")
        endif
    endfor

endprogram
