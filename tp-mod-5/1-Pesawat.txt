program Pesawat
kamus
    prim, digitSama, kelipatanTiga : boolean
    angka, nDigit, blgnTengah : integer
algoritma
    input(angka)
    nDigit <- countDigit(angka)
    blgnTengah <- ambilBilanganTengah(angka)

    prim <- isPrima(blgnTengah, 2)
    digitSama <- checkFirstLast(angka, nDigit)
    kelipatanTiga <- sumDigit(angka) mod 3 == 0

    depend on
        kelipatanTiga and digitSama and prim: output("Spesial")
        not digitSama: output("Beresiko")
        not prim: output("Biasa")
        not kelipatanTiga: output("Tidak Layak Terbang")
    enddependon
endprogram

function isPrima(n, i : integer) -> boolean
kamus
    { tidak ada }
algoritma
    if n < 2 then
        return false
    endif

    if i >= n then
        return true
    endif

    if n mod i == 0 then
        return false
    endif

    return isPrima(n, i + 1)
endfunction

function ambilBilanganTengah(n : integer) -> integer
kamus
    { tidak ada }
algoritma
    if n < 100 then
        return 0
    else
        return ambilBilanganTengah(n div 10) * 10 + (n div 10) mod 10
    endif
endfunction

function countDigit(n : integer) -> integer
kamus
    { tidak ada }
algoritma
    if n < 10 then
        return 1
    else
        return countDigit(n div 10) + 1
    endif
endfunction

function getNumber(n, i : integer) -> integer
kamus
    { tidak ada }
algoritma
    if i == 1 then
        return n mod 10
    else
        return getNumber(n div 10, i - 1)
    endif
endfunction

function checkFirstLast(n, nDigit : integer) -> boolean
kamus
    { tidak ada }
algoritma
    return getNumber(n, 1) == getNumber(n, nDigit)
endfunction

function sumDigit(n : integer) -> integer
kamus
    { tidak ada }
algoritma
    if n < 10 then
        return n
    else
        return sumDigit(n div 10) + n mod 10
    endif
endfunction
